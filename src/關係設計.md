  ## 第一部分：MySQL 資料庫架構設計 (DDL)                                                                          
                                                                                                                   
  因資料表之間存在外鍵（Foreign Key）依賴關係，建立資料表時必須遵循先建立被參照的父表、再建立引用外鍵的子表的順序：
                                                                                                                   
  1. 獨立父表：departments (部門), patients (病人), medications (藥品)                                             
  2. 第一層子表：doctors (醫生), nurses (護士), expenses (費用)                                                    
  3. 第二層子表：appointments (預約), medical_records (病歷)                                                       
  4. 關聯明細子表：prescriptions (處方明細)                                                                        
                                                                                                                   
  以下是可直接在 MySQL 執行的完整建表腳本：                                                                        
                                                                                                                   
    -- 1. 建立資料庫（使用 utf8mb4 支援完整繁體中文及特殊字元）                                                    
    CREATE DATABASE IF NOT EXISTS clinic_db

  ## 設計要點與關聯分析                                                                                            
                                                                                                                   
  1. 主鍵與自增編號：                                                                                              
      • 所有實體均採用 INT AUTO_INCREMENT 作為物理主鍵，避免原有 JSON 結構手動維護字串鍵（如 "P1",                 
      "D1"）產生衝突。若業務前端需要帶前綴的編號，可以在 Python 格式化輸出（如 f"P{patient_id:04d}"）。            
  2. 完整性約束 (Foreign Key Constraints)：                                                                        
      • 預約、病歷、費用皆與 Patient 關聯。為了防止病患重要醫療記錄被意外連鎖刪除，設定 ON DELETE RESTRICT。       
      • Prescription 隨 MedicalRecord 存在，若刪除病歷紀錄時連同處方清單一起刪除（ON DELETE CASCADE）。            
  3. 金額與時間型態選用：                                                                                          
      • 費用 amount 使用 DECIMAL(10, 2)，嚴格禁止使用浮點數 FLOAT，避免計算精度誤差。                              
      • paid_time 使用 DATETIME NULL：當為 NULL                                                                    
      即代表未繳費；繳費後填入時間戳記即代表已繳費，兼具狀態與付費時間記錄。                                       
  4. 冗餘欄位優化：                                                                                                
      • Department 表中的 doc_count 與 nurse_count 是冗餘衍生欄位，在 MySQL 中可隨時用 SELECT COUNT(*) FROM doctors
      WHERE dept_id = ? 動態取得，保持資料一致性。

  ### 步驟 3：建立資料庫連線工具模組 (src/db.py)                                                                   
                                                                                                                   
  建立一個獨立的模組負責管理連線，並使用 DictCursor（查詢結果自動包裝為 Python 字典，結構與目前前端 Svelte 接收的  
  JSON 格式無縫對接）：                                                                                            
                                                                                                                   
    # src/db.py                                                                                                    
    import os                                                                                                      
    import pymysql                                                                                                 
    from pymysql.cursors import DictCursor                                                                         
    from dbutils.pooled_db import PooledDB                                                                         
    from dotenv import load_dotenv                                                                                 
    from contextlib import contextmanager                                                                          
                                                                                                                   
    load_dotenv()                                                                                                  
                                                                                                                   
    # 初始化連線池                                                                                                 
    pool = PooledDB(                                                                                               
        creator=pymysql,                                                                                           
        maxconnections=10,                                                                                         
        mincached=2,                                                                                               
        maxcached=5,                                                                                               
        blocking=True,                                                                                             
        host=os.getenv("DB_HOST", "127.0.0.1"),                                                                    
        port=int(os.getenv("DB_PORT", 3306)),                                                                      
        user=os.getenv("DB_USER", "root"),                                                                         
        password=os.getenv("DB_PASSWORD", ""),                                                                     
        database=os.getenv("DB_NAME", "clinic_db"),                                                                
        charset=os.getenv("DB_CHARSET", "utf8mb4"),                                                                
        cursorclass=DictCursor,                                                                                    
        autocommit=False  # 由我們手動控制交易 (Transaction)                                                       
    )                                                                                                              
                                                                                                                   
    @contextmanager                                                                                                
    def get_db():                                                                                                  
        """提供 context manager 方便安全獲取連線與自動關閉/回滾"""                                                 
        conn = pool.connection()                                                                                   
        cursor = conn.cursor()                                                                                     
        try:                                                                                                       
            yield cursor                                                                                           
            conn.commit()                                                                                          
        except Exception as e:                                                                                     
            conn.rollback()                                                                                        
            raise e                                                                                                
        finally:                                                                                                   
            cursor.close()                                                                                         
            conn.close()
